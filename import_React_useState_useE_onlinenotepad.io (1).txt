const generateMoreEmails = (startIndex, count) => {

  const newEmails = [];

  const senders = ["Trendyol", "Yemeksepeti", "Google", "Apple", "Spotify", "Zoom", "Slack", "Müşteri A", "Müşteri B"];

  const subjects = ["Siparişiniz alındı", "Güvenlik uyarısı", "Faturanız hazır", "Toplantı hatırlatması", "Yeni giriş yapıldı", "Haftalık rapor"];

  

  for (let i = 0; i < count; i++) {

    const randomSender = senders[Math.floor(Math.random() * senders.length)];

    const randomSubject = subjects[Math.floor(Math.random() * subjects.length)];

    const uniqueId = `more-${Date.now()}-${i}-${Math.random().toString(36).substr(2, 9)}`;



    newEmails.push({

      id: uniqueId, 

      folderId: 'inbox',

      from: randomSender,

      email: `info@${randomSender.toLowerCase()}.com`,

      subject: `${randomSubject} #${startIndex + i}`,

      body: "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Sed do eiusmod tempor incididunt ut labore et dolore magna aliqua.",

      date: `${Math.floor(Math.random() * 28) + 1} Şub`,

      read: Math.random() > 0.3,

      starred: false,

      isPinned: false,

      labels: Math.random() > 0.7 ? ['İş'] : []

    });

  }

  return newEmails;

};



const getAvatarColor = (name) => {

  if (!name) return MATTE_AVATAR_COLORS[0];

  const charCode = name.charCodeAt(0);

  const index = charCode % MATTE_AVATAR_COLORS.length;

  return MATTE_AVATAR_COLORS[index];

};

  // Etiketler

  const [availableLabels, setAvailableLabels] = useState(INITIAL_LABELS);



  // Infinite Scroll & Selection States

  const [loadingMore, setLoadingMore] = useState(false);

  const [selectedEmailIds, setSelectedEmailIds] = useState([]);



  const [settings, setSettings] = useState({

    notifications: true,

    syncFrequency: 'push',

  });



  const currentAccount = accounts.find(a => a.id === currentAccountId) || {};



  useEffect(() => {

    const savedAccounts = localStorage.getItem('kaydet_accounts');

    if (savedAccounts) {

      const parsedAccounts = JSON.parse(savedAccounts);

      if (parsedAccounts.length > 0) {

        setAccounts(parsedAccounts);

        setCurrentAccountId(parsedAccounts[0].id);

        setCurrentScreen('inbox');

      } else {

        setCurrentScreen('login');

      }

    } else {

      setCurrentScreen('login');

    }

  }, []);



  useEffect(() => {

    if (accounts.length > 0) {

        localStorage.setItem('kaydet_accounts', JSON.stringify(accounts));

    }

  }, [accounts]);

  

  // FİLTRELEME & ARAMA MANTIĞI

  const filteredEmails = emails.filter(e => {

    // 1. Klasör Filtresi

    let matchesFolder = false;

    if (currentFolder === 'pinned') matchesFolder = e.isPinned;

    else matchesFolder = e.folderId === currentFolder;



    // 2. Arama Filtresi

    let matchesSearch = true;

    if (searchQuery.trim() !== '') {

        const q = searchQuery.toLowerCase();

        matchesSearch = e.subject.toLowerCase().includes(q) || 

                        e.from.toLowerCase().includes(q) || 

                        e.body.toLowerCase().includes(q);

    }



    return matchesFolder && matchesSearch;

  });



  const pinnedEmailsCount = emails.filter(e => e.isPinned).length;

  const isSelectionMode = selectedEmailIds.length > 0;



  const getFolderCount = (folderId) => {

    if (folderId === 'pinned') return pinnedEmailsCount;

    if (folderId === 'drafts') return emails.filter(e => e.folderId === 'drafts').length;

    return emails.filter(e => e.folderId === folderId && !e.read).length;

  };



  const handleLoginSuccess = (credentials) => {

    const newAccount = {

      id: Date.now(),

      email: credentials.email,

      name: credentials.email.split('@')[0],

      signature: `\n\n--\n${credentials.email.split('@')[0]}\nSent from KAYDET Mail`,

      settings: { ...credentials }

    };

    const updatedAccounts = [...accounts, newAccount];

    setAccounts(updatedAccounts);

    setCurrentAccountId(newAccount.id);

    setCurrentScreen('inbox');

  };



  const handleLogout = () => {

    const remainingAccounts = accounts.filter(a => a.id !== currentAccountId);

    setAccounts(remainingAccounts);

    if (remainingAccounts.length > 0) {

        setCurrentAccountId(remainingAccounts[0].id);

    } else {

        localStorage.removeItem('kaydet_accounts');

        setCurrentAccountId(null);

        setCurrentScreen('login');

    }

  };



  const handleSendMail = (mailData) => {

    let updatedEmails = emails;

    if (mailData.id) {

        updatedEmails = emails.filter(e => e.id !== mailData.id);

    }

    const newMailId = `sent-${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;

    setEmails([{ 

        id: newMailId, 

        folderId: 'sent', 

        from: "Ben", 

        ...mailData, 

        date: "Şimdi", 

        read: true, 

        isPinned: false, 

        labels: [] 

    }, ...updatedEmails]);

    setCurrentScreen('inbox');

  };



  const handleSaveDraft = (mailData) => {

    let updatedEmails = emails;

    const draftId = mailData.id || `draft-${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;



    if (mailData.id) {

        updatedEmails = emails.map(e => e.id === mailData.id ? {

             ...e,

             ...mailData,

             date: "Taslak",

             folderId: 'drafts',

             read: true

        } : e);

    } else {

        updatedEmails = [{ 

            id: draftId, 

            folderId: 'drafts', 

            from: "Taslak", 

            email: "Ben",

            ...mailData, 

            date: "Taslak", 

            read: true, 

            isPinned: false, 

            labels: [] 

        }, ...emails];

    }

    setEmails(updatedEmails);

    setCurrentScreen('inbox');

  };



  const handleDeleteMail = (id) => {

    setEmails(emails.map(e => e.id === id ? { ...e, folderId: 'trash', isPinned: false } : e));

    if (currentScreen === 'detail') setCurrentScreen('inbox');

  };



  const handleTogglePin = (id) => {

    setEmails(emails.map(e => e.id === id ? { ...e, isPinned: !e.isPinned } : e));

  };



  const handleAddLabel = (id, labelName) => {

    const labelObj = availableLabels.find(l => l.name === labelName);

    if(!labelObj) return; 

    setEmails(emails.map(e => {

      if (e.id === id) {

        if (e.labels && e.labels.includes(labelName)) return e; 

        return { ...e, labels: [...(e.labels || []), labelName] };

      }

      return e;

    }));

  };



  const toggleSelection = (id) => {

    if (selectedEmailIds.includes(id)) {

        setSelectedEmailIds(selectedEmailIds.filter(sid => sid !== id));

    } else {

        setSelectedEmailIds([...selectedEmailIds, id]);

    }

  };



  const handleSelectAll = () => {

      if (selectedEmailIds.length === filteredEmails.length) {

          setSelectedEmailIds([]);

      } else {

          setSelectedEmailIds(filteredEmails.map(e => e.id));

      }

  };



  const handleBulkAction = (action) => {

      if (action === 'delete') {

          setEmails(emails.map(e => selectedEmailIds.includes(e.id) ? { ...e, folderId: 'trash' } : e));

      } else if (action === 'read') {

          setEmails(emails.map(e => selectedEmailIds.includes(e.id) ? { ...e, read: true } : e));

      } else if (action === 'unread') {

          setEmails(emails.map(e => selectedEmailIds.includes(e.id) ? { ...e, read: false } : e));

      } else if (action === 'pin') {

          setEmails(emails.map(e => selectedEmailIds.includes(e.id) ? { ...e, isPinned: !e.isPinned } : e));

      } else if (action === 'archive') {

          alert(`${selectedEmailIds.length} mail arşivlendi.`);

      }

      setSelectedEmailIds([]); 

  };



  const handleNavigateMail = (direction) => {

    const currentList = filteredEmails;

    const currentIndex = currentList.findIndex(e => e.id === selectedEmail.id);

    

    let newIndex = -1;

    if (direction === 'prev') newIndex = currentIndex - 1;

    if (direction === 'next') newIndex = currentIndex + 1;



    if (newIndex >= 0 && newIndex < currentList.length) {

      setSelectedEmail(currentList[newIndex]);

    }

  };



  const handleOpenMail = (mail) => {

      if (isSelectionMode) {

          toggleSelection(mail.id);

          return;

      }

      if (mail.folderId === 'drafts') {

          setComposeData({

              id: mail.id,

              to: mail.to || '',

              cc: mail.cc || '',

              bcc: mail.bcc || '',

              subject: mail.subject || '',

              body: mail.body || ''

          });

          setCurrentScreen('compose');

      } else {

          setSelectedEmail(mail);

          setCurrentScreen('detail');

      }

  };



  const handleScroll = (e) => {

    const bottom = e.target.scrollHeight - e.target.scrollTop <= e.target.clientHeight + 50; 

    if (bottom && !loadingMore && currentFolder === 'inbox' && !searchQuery) { 

      loadMoreEmails();

    }

  };



  const loadMoreEmails = () => {

    setLoadingMore(true);

    setTimeout(() => {

      const newEmails = generateMoreEmails(emails.length + 100, 20); 

      setEmails(prevEmails => [...prevEmails, ...newEmails]);

      setLoadingMore(false);

    }, 1000);

  };



  if (currentScreen === 'loading') {

      return <div className="flex justify-center items-center h-screen bg-white"><RefreshCw className="animate-spin text-black" /></div>;

  }



  return (

    <div className="flex justify-center bg-gray-100 min-h-screen font-sans text-black">

       <div className="w-full max-w-md bg-white shadow-2xl h-screen relative overflow-hidden flex flex-col border-x border-gray-200">

          {(currentScreen === 'login' || currentScreen === 'add_account') && 

            <LoginScreen 

              onLogin={handleLoginSuccess} 

              isAddingAccount={currentScreen === 'add_account'}

            />

          }

          {currentScreen !== 'login' && currentScreen !== 'add_account' && (

             <>

                <Sidebar 

                  isSidebarOpen={isSidebarOpen} setIsSidebarOpen={setIsSidebarOpen}

                  currentAccount={currentAccount} accounts={accounts}

                  currentAccountId={currentAccountId} setCurrentAccountId={setCurrentAccountId}

                  setCurrentScreen={setCurrentScreen} currentFolder={currentFolder}

                  setCurrentFolder={setCurrentFolder} getFolderCount={getFolderCount}

                  availableLabels={availableLabels}

                />

                

                {currentScreen === 'inbox' && 

                  <InboxScreen 

                    isSelectionMode={isSelectionMode} selectedEmailIds={selectedEmailIds} 

                    setSelectedEmailIds={setSelectedEmailIds} handleSelectAll={handleSelectAll}

                    isSearchOpen={isSearchOpen} setIsSearchOpen={setIsSearchOpen}

                    searchQuery={searchQuery} setSearchQuery={setSearchQuery}

                    setIsSidebarOpen={setIsSidebarOpen} currentFolder={currentFolder}

                    handleScroll={handleScroll} pinnedEmailsCount={pinnedEmailsCount}

                    loadingMore={loadingMore} filteredEmails={filteredEmails}

                    handleOpenMail={handleOpenMail} toggleSelection={toggleSelection}

                    getAvatarColor={getAvatarColor} availableLabels={availableLabels}

                    currentAccount={currentAccount} setComposeData={setComposeData}

                    setCurrentScreen={setCurrentScreen} handleBulkAction={handleBulkAction}

                    setCurrentFolder={setCurrentFolder}

                  />

                }

                

                {currentScreen === 'detail' && 

                  <DetailScreen 

                     setCurrentScreen={setCurrentScreen} handleNavigateMail={handleNavigateMail}

                     hasPrev={filteredEmails.findIndex(e => e.id === selectedEmail?.id) > 0} 

                     hasNext={filteredEmails.findIndex(e => e.id === selectedEmail?.id) < filteredEmails.length - 1}

                     selectedEmail={selectedEmail} handleTogglePin={handleTogglePin}

                     handleAddLabel={handleAddLabel} handleDeleteMail={handleDeleteMail}

                     setComposeData={setComposeData} availableLabels={availableLabels}

                  />

                }

                

                {currentScreen === 'compose' && 

                  <ComposeScreen 

                    composeData={composeData} currentAccount={currentAccount}

                    setCurrentScreen={setCurrentScreen} handleSendMail={handleSendMail}

                    handleSaveDraft={handleSaveDraft}

                  />

                }

                

                {currentScreen === 'settings' && 

                  <SettingsScreen 

                    setCurrentScreen={setCurrentScreen} availableLabels={availableLabels}

                    setAvailableLabels={setAvailableLabels} currentAccount={currentAccount}

                    settings={settings} setSettings={setSettings} handleLogout={handleLogout}

                  />

                }

             </>

          )}

       </div>

       <style>{`.animate-fade-in { animation: fadeIn 0.3s ease-out; } @keyframes fadeIn { from { opacity: 0; } to { opacity: 1; } }`}</style>

    </div>

  );

}
